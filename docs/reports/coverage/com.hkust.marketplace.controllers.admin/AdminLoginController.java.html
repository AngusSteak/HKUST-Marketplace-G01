package com.hkust.marketplace.controllers.admin;

import com.hkust.marketplace.models.User;
import com.hkust.marketplace.models.UserRole;
import com.hkust.marketplace.utils.DummyDataService;
import com.hkust.marketplace.utils.LoginGate;
import com.hkust.marketplace.utils.PasswordVisibility;
import com.hkust.marketplace.utils.SceneManager;
import com.hkust.marketplace.utils.SessionManager;
import javafx.fxml.FXML;
import javafx.scene.control.Label;
import javafx.scene.control.PasswordField;
import javafx.scene.control.TextField;
import javafx.scene.layout.HBox;

/** Handles the admin login form and opens the admin dashboard. */
public class AdminLoginController {

    @FXML private TextField usernameField;
    @FXML private PasswordField passwordField;
    @FXML private HBox passwordRow;
    @FXML private Label errorLabel;

    @FXML
    private void initialize() {
        PasswordVisibility.attach(passwordField, passwordRow);
    }

    @FXML
    private void handleLogin() {
        String username = usernameField.getText().trim();
        String password = passwordField.getText();
        errorLabel.setVisible(false);
        errorLabel.setManaged(false);

        if (username.isEmpty() || password.isEmpty()) {
            showError("Please enter credentials.");
            return;
        }
        User user = DummyDataService.getInstance().users().find(username);
        switch (LoginGate.check(user, UserRole.ADMIN, password)) {
            case USER_NOT_FOUND, WRONG_PORTAL -> showError("Invalid admin credentials.");
            case WRONG_PASSWORD -> showError("Incorrect password.");
            case DELETED -> showError("This account has been deleted and cannot log in.");
            case DISABLED -> showError("This account has been deactivated. Contact admin.");
            case OK -> {
                SessionManager.login(user);
                SceneManager.showAdminDashboard();
            }
        }
    }

    private void showError(String msg) {
        errorLabel.setText(msg);
        errorLabel.setVisible(true);
        errorLabel.setManaged(true);
    }

    @FXML private void goWelcome() { SceneManager.showWelcome(); }
}
